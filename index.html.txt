    <script>
        const canvas = document.getElementById('particle-canvas');
        const ctx = canvas.getContext('2d');
        let particles = [];

        function init() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            particles = Array.from({ length: 50 }, () => ({
                x: Math.random() * canvas.width,
                y: Math.random() * canvas.height,
                s: Math.random() * 4 + 1,
                vx: (Math.random() - 0.5) * 0.4,
                vy: (Math.random() - 0.5) * 0.4,
                c: Math.random() > 0.5 ? '#94e524' : '#ff007f',
                o: Math.random() * 0.2 + 0.05
            }));
        }

        function draw() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            particles.forEach(p => {
                p.x += p.vx; p.y += p.vy;
                if (p.x < 0 || p.x > canvas.width) p.vx *= -1;
                if (p.y < 0 || p.y > canvas.height) p.vy *= -1;
                ctx.globalAlpha = p.o;
                ctx.fillStyle = p.c;
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.s / 2, 0, Math.PI * 2);
                ctx.fill();
            });

            ctx.globalAlpha = 0.03;
            ctx.strokeStyle = '#94e524';
            ctx.lineWidth = 0.5;
            for (let i = 0; i < particles.length; i++) {
                for (let j = i + 1; j < particles.length; j++) {
                    const dx = particles[i].x - particles[j].x;
                    const dy = particles[i].y - particles[j].y;
                    const dist = Math.sqrt(dx * dx + dy * dy);
                    if (dist < 120) {
                        ctx.globalAlpha = 0.03 * (1 - dist / 120);
                        ctx.beginPath();
                        ctx.moveTo(particles[i].x, particles[i].y);
                        ctx.lineTo(particles[j].x, particles[j].y);
                        ctx.stroke();
                    }
                }
            }

            requestAnimationFrame(draw);
        }

        window.addEventListener('resize', init);
        init();
        draw();

        ];

        let idx = 0, char = 0, isDeleting = false;

        function type() {
            const cur = texts[idx];
            const langEl = document.getElementById('lang-sub');
            const el = document.getElementById('typewriter');

            langEl.innerText = cur.l;
            langEl.style.opacity = '1';
            const currentText = isDeleting ? cur.t.slice(0, char--) : cur.t.slice(0, char++);
            el.innerText = currentText;

            let speed = isDeleting ? 40 : 80;

            if (!isDeleting && char > cur.t.length) {
                isDeleting = true;
                langEl.style.opacity = '0.5';
                speed = 2500;
            } else if (isDeleting && char < 0) {
                isDeleting = false;
                idx = (idx + 1) % texts.length;
                langEl.style.opacity = '1';
                speed = 400;
            }

            setTimeout(type, speed);
        }

        function timer() {
            const target = new Date("May 21, 2026 14:00:00").getTime();
            const diff = target - new Date().getTime();

            if (diff > 0) {
                document.getElementById('days').innerText = String(Math.floor(diff / 86400000)).padStart(2, '0');
                document.getElementById('hours').innerText = String(Math.floor((diff % 86400000) / 3600000)).padStart(2, '0');
                document.getElementById('minutes').innerText = String(Math.floor((diff % 3600000) / 60000)).padStart(2, '0');
                document.getElementById('seconds').innerText = String(Math.floor((diff % 60000) / 1000)).padStart(2, '0');
            } else {
                document.getElementById('days').innerText = "00";
                document.getElementById('hours').innerText = "00";
                document.getElementById('minutes').innerText = "00";
                document.getElementById('seconds').innerText = "00";
            }
        }

        setInterval(timer, 1000);
        timer();
        type();
